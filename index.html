form.addEventListener('submit', function(e){ e.preventDefault(); submitBtn.disabled = true; statusEl.textContent = 'Submitting…'; statusEl.className = 'form-status'; var data = { name: form.name.value.trim(), email: form.email.value.trim(), student_id: form.student_id.value.trim(), major: form.major.value.trim(), year_level: form.year_level.value, committee_choice: form.committee_choice.value, why_join: form.why_join.value.trim() }; fetch(SUPABASE_URL + '/rest/v1/Applications', { method: 'POST', headers: { 'Content-Type': 'application/json', 'apikey': SUPABASE_KEY, 'Authorization': 'Bearer ' + SUPABASE_KEY, 'Prefer': 'return=minimal' }, body: JSON.stringify(data) }).then(function(res){ if (res.ok) { statusEl.textContent = 'Application sent — thank you!'; statusEl.className = 'form-status ok'; form.reset(); setTimeout(closeApplyModal, 1600); } else { return res.text().then(function(t){ throw new Error('HTTP ' + res.status + ': ' + t); }); } }).catch(function(err){ statusEl.textContent = err.message; // TEMP: shows the real error so we can diagnose it statusEl.className = 'form-status err'; console.error(err); }).finally(function(){ submitBtn.disabled = false; }); });
